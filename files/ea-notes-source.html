% --------------------------------------------------------------------------
% Theorem-like environments
% --------------------------------------------------------------------------

% One skip above each environment's name and below the bottom of its box,
% so everything sits on a common vertical grid.
\newskip\ea@skip
\ea@skip=\medskipamount

% Fully boxed, for theorem/lemma/etc.
\mdfdefinestyle{thmstyle}{%
	topline=true,%
	bottomline=true,%
	leftline=true,%
	rightline=true,%
	linewidth=1pt,%
	linecolor=gray-9,%
	innertopmargin=8pt,%
	innerbottommargin=8pt,%
	innerleftmargin=8pt,%
	innerrightmargin=8pt,%
	skipabove=\smallskipamount,%
	skipbelow=\ea@skip,%
	nobreak=false%
}

% Just a rule down the left, for definition/question.
\mdfdefinestyle{defstyle}{%
	topline=false,%
	bottomline=false,%
	rightline=false,%
	leftline=true,%
	linewidth=0.75ex,%
	linecolor=gray-9,%
	innertopmargin=0pt,%
	innerbottommargin=0pt,%
	innerleftmargin=8pt,%
	innerrightmargin=0pt,%
	skipabove=\ea@skip,%
	skipbelow=\ea@skip,%
	nobreak=false%
}

% Each numbers on its own; definition and question stay unnumbered.
\newcounter{theorem}
\newcounter{lemma}
\newcounter{proposition}
\newcounter{corollary}
\newcounter{conjecture}

% Heading for the boxed environments: " " with an optional title.
% \needspace keeps the heading from being orphaned above its box at a
% page break.
\newcommand{\ea@thmheader}[3]{%
	\def\ea@thmtitle{#2}%
	\needspace{6\baselineskip}%
	\par\vspace{\ea@skip}\noindent
	{\large\sffamily\bfseries\color{gray-9}#1~\csname the#3\endcsname}%
	\ifx\ea@thmtitle\@empty\else
		\hspace{2em}{\large\normalfont\sffamily\color{gray-9}#2}%
	\fi
	\par\nobreak\smallskip
}

\newenvironment{theorem}[1][]{%
	\refstepcounter{theorem}\ea@thmheader{Theorem}{#1}{theorem}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{lemma}[1][]{%
	\refstepcounter{lemma}\ea@thmheader{Lemma}{#1}{lemma}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{proposition}[1][]{%
	\refstepcounter{proposition}\ea@thmheader{Proposition}{#1}{proposition}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{corollary}[1][]{%
	\refstepcounter{corollary}\ea@thmheader{Corollary}{#1}{corollary}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{conjecture}[1][]{%
	\refstepcounter{conjecture}\ea@thmheader{Conjecture}{#1}{conjecture}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{definition}[1][]{%
	\def\definition@title{#1}%
	\begin{mdframed}[style=defstyle]%
	\noindent{\sffamily\bfseries\color{gray-9}%
		Definition%
		\ifx\definition@title\@empty\else:\hspace{1ex}\definition@title\fi}%
	\par\smallskip\noindent
}{\end{mdframed}}

\newenvironment{question}[1][]{%
	\def\question@title{#1}%
	\begin{mdframed}[style=defstyle]%
	\noindent{\sffamily\bfseries\color{gray-9}%
		Question%
		\ifx\question@title\@empty\else:\hspace{1ex}\question@title\fi}%
	\par\smallskip\noindent
}{\end{mdframed}}

% Examples run in the text rather than in a box, with a small-caps label.
\newcounter{example}

\newenvironment{example}[1][]{%
	\def\ea@exampletitle{#1}%
	\refstepcounter{example}%
	\par\vspace{\ea@skip}\noindent
	{\normalsize\sffamily\bfseries\color{gray-9}EXAMPLE~\theexample}%
	\ifx\ea@exampletitle\@empty\else
		\hspace{2em}{\normalsize\normalfont\sffamily\color{gray-9}\ea@exampletitle}%
	\fi
	\par\smallskip\noindent\ignorespaces
}{\par\vspace{\ea@skip}}

% Proof: the optional argument (e.g. "of \cref{thm:foo}") sits inline next
% to the PROOF label in the same style.
\newenvironment{proof}[1][]{%
	\def\ea@prooftitle{#1}%
	\par\vspace{\ea@skip}\noindent
	\textbf{\sffamily PROOF\ifx\ea@prooftitle\@empty\else\ \ea@prooftitle\fi}%
	\quad\ignorespaces
}{\hfill$\blacksquare$\par\vspace{\ea@skip}}

% --------------------------------------------------------------------------
% Page layout
% --------------------------------------------------------------------------

% Running header: document title on odd pages, current section on even
% pages, page number at the right, rule on every page.  \maketitle switches
% the title page to an empty style.
\providecommand{\sectionmark}[1]{}
\renewcommand{\sectionmark}[1]{\markright{#1}}

\newcommand{\ea@runningtitle}{\ifodd\value{page}\@title\else\rightmark\fi}

\pagestyle{fancy}
\fancyhf{}
\fancyhead[C]{\small\sffamily\ea@runningtitle}
\fancyhead[R]{\small\sffamily\bfseries\thepage}
\renewcommand{\headrulewidth}{1pt}
\setlength{\headheight}{14pt}

\renewcommand{\maketitle}{%
	\thispagestyle{empty}%
	\noindent{\Huge\sffamily\color{black}\@title}\par
	\vspace{2\baselineskip}%
}

% Headings are unnumbered throughout.
\setcounter{secnumdepth}{-2}

\titleformat{\section}[hang]
{\Large\sffamily\bfseries\color{black}}
{\thesection\enspace}
{0pt}
{}
[\vspace{-12pt}\rule{\linewidth}{1pt}]
\titlespacing*{\section}{0pt}{1.5\baselineskip}{0.75\baselineskip}

\titleformat{\subsection}[hang]
{\large\sffamily\bfseries\color{black}}
{\thesubsection\enspace}
{0pt}
{}
[\vspace{0pt}]
\titlespacing*{\subsection}{0pt}{\baselineskip}{0.5\baselineskip}

% Subsubsections match, one size down, and run into the following text.
\titleformat{\subsubsection}[runin]
{\normalsize\sffamily\bfseries\color{black}}
{\thesubsubsection\enspace}
{0pt}
{}
\titlespacing*{\subsubsection}{0pt}{0.75\baselineskip}{1em}
