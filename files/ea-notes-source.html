% --- theorem-like environments ---------------------------------------------

% frame used by theorem/lemma/proposition/corollary/conjecture
\mdfdefinestyle{thmstyle}{%
	topline=true,%
	bottomline=true,%
	leftline=true,%
	rightline=true,%
	linewidth=1pt,%
	linecolor=gray-9,%
	innertopmargin=8pt,%
	innerbottommargin=8pt,%
	innerleftmargin=8pt,%
	innerrightmargin=8pt,%
	skipabove=\smallskipamount,%
	skipbelow=\bigskipamount%
}

% lighter-weight style for definition/question - just a rule on the left
\mdfdefinestyle{defstyle}{%
	topline=false,%
	bottomline=false,%
	rightline=false,%
	leftline=true,%
	linewidth=0.75ex,%
	linecolor=gray-9,%
	innertopmargin=0pt,%
	innerbottommargin=0pt,%
	innerleftmargin=8pt,%
	innerrightmargin=0pt,%
	skipabove=\bigskipamount,%
	skipbelow=\bigskipamount%
}

% theorem/lemma/proposition/corollary/conjecture share one counter
\newcounter{theorem}

\newcommand{\ea@thmheader}[2]{%
	\def\ea@thmtitle{#2}%
	\par\bigskip\noindent
	{\large\sffamily\bfseries\color{gray-9}#1~\thetheorem}%
	\ifx\ea@thmtitle\@empty\else
		\hspace{2em}{\large\normalfont\sffamily\color{gray-9}#2}%
	\fi
	\par\nobreak\smallskip
}

\newenvironment{theorem}[1][]{%
	\refstepcounter{theorem}\ea@thmheader{Theorem}{#1}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{lemma}[1][]{%
	\refstepcounter{theorem}\ea@thmheader{Lemma}{#1}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{proposition}[1][]{%
	\refstepcounter{theorem}\ea@thmheader{Proposition}{#1}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{corollary}[1][]{%
	\refstepcounter{theorem}\ea@thmheader{Corollary}{#1}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{conjecture}[1][]{%
	\refstepcounter{theorem}\ea@thmheader{Conjecture}{#1}%
	\begin{mdframed}[style=thmstyle]\itshape
}{\end{mdframed}}

\newenvironment{definition}[1][]{%
	\def\definition@title{#1}%
	\begin{mdframed}[style=defstyle]%
	\noindent{\sffamily\bfseries\color{gray-9}%
		Definition%
		\ifx\definition@title\@empty\else:\hspace{1ex}\definition@title\fi}%
	\par\smallskip\noindent
}{\end{mdframed}}

\newenvironment{question}[1][]{%
	\def\question@title{#1}%
	\begin{mdframed}[style=defstyle]%
	\noindent{\sffamily\bfseries\color{gray-9}%
		Question%
		\ifx\question@title\@empty\else:\hspace{1ex}\question@title\fi}%
	\par\smallskip\noindent
}{\end{mdframed}}

\newcounter{example}

\newenvironment{example}[1][]{%
	\def\ea@exampletitle{#1}%
	\refstepcounter{example}%
	\par\bigskip\noindent
	{\normalsize\sffamily\bfseries\color{gray-9}EXAMPLE~\theexample}%
	\ifx\ea@exampletitle\@empty\else
		\hspace{2em}{\normalsize\normalfont\sffamily\color{gray-9}\ea@exampletitle}%
	\fi
	\par\smallskip\noindent\ignorespaces
}{\par\bigskip}

\newenvironment{proof}{%
	\par\bigskip\noindent\textbf{\sffamily PROOF}\quad\ignorespaces
}{\hfill$\blacksquare$\par\bigskip}
